    Requests failing with “Failed to fetch”? Use a proxy.

    Adobe’s API does not send CORS headers, so browsers block direct calls, and the job’s Location header is hidden from scripts. A small local proxy fixes both and keeps your calls off third-party servers. Save this as proxy.mjs, run node proxy.mjs (Node 18+), then enter http://localhost:8787 above. Limit the allowed origin before using it anywhere but your own machine.

    import http from 'node:http';
const cors = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': '*',
  'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
  'Access-Control-Expose-Headers': 'Location,x-request-id'
};
http.createServer(async (req, res) => {
  if (req.method === 'OPTIONS') { res.writeHead(204, cors); return res.end(); }
  const chunks = []; for await (const c of req) chunks.push(c);
  const h = {};
  for (const k of ['content-type', 'authorization', 'x-api-key'])
    if (req.headers[k]) h[k] = req.headers[k];
  const r = await fetch('https://pdf-services.adobe.io' + req.url, {
    method: req.method, headers: h,
    body: chunks.length ? Buffer.concat(chunks) : undefined });
  const out = { ...cors, 'content-type': r.headers.get('content-type') || 'application/json' };
  const loc = r.headers.get('location'); if (loc) out.Location = loc;
  res.writeHead(r.status, out);
  res.end(Buffer.from(await r.arrayBuffer()));
}).listen(8787, () => console.log('Proxy on http://localhost:8787'));
